[CmdletBinding()]
param([int]$IntervalSeconds = 30)

$ErrorActionPreference = "Stop"
$ProjectRoot = (Resolve-Path (Join-Path $PSScriptRoot "..\..")).Path
$LogDir = Join-Path $ProjectRoot "logs"
New-Item -ItemType Directory -Path $LogDir -Force | Out-Null
$LogFile = Join-Path $LogDir "networklab-watchdog.log"

$PythonCommand = Get-Command python -ErrorAction SilentlyContinue
if (-not $PythonCommand) {
    throw "Python is niet gevonden in PATH. Installeer Python of herstel PATH."
}
$Python = $PythonCommand.Source

$Apps = @(
    @{ Name="CLIENT"; Port=8510; Url="http://127.0.0.1:8510/"; Directory=(Join-Path $ProjectRoot "apps\client"); Kind="static" },
    @{ Name="MGMT"; Port=8511; Url="http://127.0.0.1:8511/"; Directory=(Join-Path $ProjectRoot "apps\mgmt"); Kind="static" },
    @{ Name="INFRA"; Port=8512; Url="http://127.0.0.1:8512/"; Directory=(Join-Path $ProjectRoot "apps\infra"); Kind="infra" }
)

function Write-WatchLog([string]$Message) {
    $Line = "$(Get-Date -Format 'yyyy-MM-dd HH:mm:ss') $Message"
    Add-Content -LiteralPath $LogFile -Value $Line -Encoding UTF8
    Write-Host $Line
}

function Test-App([string]$Url) {
    try {
        $Response = Invoke-WebRequest -Uri $Url -Method Get -UseBasicParsing -TimeoutSec 4
        return ([int]$Response.StatusCode -eq 200)
    } catch {
        return $false
    }
}

function Start-App($App) {
    if (-not (Test-Path -LiteralPath $App.Directory -PathType Container)) {
        Write-WatchLog "ERROR $($App.Name): map ontbreekt: $($App.Directory)"
        return
    }

    $Listener = Get-NetTCPConnection -State Listen -LocalPort $App.Port -ErrorAction SilentlyContinue
    if ($Listener) {
        Write-WatchLog "WARNING $($App.Name): poort $($App.Port) is bezet, maar de URL reageert niet. Geen proces gestopt."
        return
    }

    $OutLog = Join-Path $LogDir ("networklab-" + $App.Name.ToLowerInvariant() + ".out.log")
    $ErrLog = Join-Path $LogDir ("networklab-" + $App.Name.ToLowerInvariant() + ".err.log")

    if ($App.Kind -eq "infra") {
        $ServerFile = Join-Path $App.Directory "server.py"
        if (-not (Test-Path -LiteralPath $ServerFile -PathType Leaf)) {
            Write-WatchLog "ERROR $($App.Name): server.py ontbreekt."
            return
        }
        $Arguments = '"' + $ServerFile + '"'
    } else {
        $Arguments = '-m http.server ' + $App.Port +
            ' --bind 127.0.0.1 --directory "' + $App.Directory + '"'
    }

    try {
        $null = Start-Process -FilePath $Python -ArgumentList $Arguments `
            -WorkingDirectory $ProjectRoot -WindowStyle Hidden `
            -RedirectStandardOutput $OutLog -RedirectStandardError $ErrLog -PassThru
        Start-Sleep -Seconds 3

        if (Test-App $App.Url) {
            Write-WatchLog "RECOVERED $($App.Name): HTTP 200 op poort $($App.Port)."
        } else {
            Write-WatchLog "ERROR $($App.Name): gestart, maar nog geen HTTP 200. Controleer $ErrLog"
        }
    } catch {
        Write-WatchLog "ERROR $($App.Name): starten mislukt: $($_.Exception.Message)"
    }
}

Write-WatchLog "WATCHDOG gestart. Controle-interval: $IntervalSeconds seconden."
Write-WatchLog "Projectmap: $ProjectRoot"

while ($true) {
    foreach ($App in $Apps) {
        if (Test-App $App.Url) {
            Write-Host "$(Get-Date -Format 'HH:mm:ss') OK $($App.Name)"
        } else {
            Write-WatchLog "DOWN $($App.Name): geen HTTP 200; herstelpoging."
            Start-App $App
        }
    }

    Start-Sleep -Seconds $IntervalSeconds
}
